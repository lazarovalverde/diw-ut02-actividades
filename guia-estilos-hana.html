CSS tokens — copia esto en tu style.css

Pega este bloque al inicio de tu archivo CSS, antes de cualquier otra regla. Nunca uses valores sueltos: siempre referencia las variables.

:root { /* ── Colores ── */ --color-brand: hsl(0 70% 35%); --color-brand-dark: hsl(0 70% 25%); --color-brand-lt: hsl(0 70% 96%); --color-accent: hsl(35 90% 52%); --color-accent-lt: hsl(35 90% 94%); --color-ink: hsl(220 15% 12%); --color-muted: hsl(220 8% 48%); --color-subtle: hsl(220 8% 68%); --color-bg: hsl(35 20% 97%); --color-surface: hsl(0 0% 100%); --color-border: hsl(35 15% 88%); --color-border-strong: hsl(35 15% 75%); /* ── Tipografía ── */ --font-display: 'Playfair Display', Georgia, serif; --font-body: 'DM Sans', system-ui, sans-serif; /* ── Escala tipográfica ── */ --text-xs: 0.75rem; --text-sm: 0.875rem; --text-base: 1rem; --text-lg: 1.125rem; --text-xl: 1.25rem; --text-2xl: 1.5rem; --text-3xl: 1.875rem; --text-4xl: 2.25rem; --text-5xl: 3rem; /* ── Line-height ── */ --lh-tight: 1.2; --lh-snug: 1.4; --lh-body: 1.7; --lh-ui: 1; /* ── Espaciado ── */ --space-1: 0.25rem; --space-2: 0.5rem; --space-3: 0.75rem; --space-4: 1rem; --space-5: 1.25rem; --space-6: 1.5rem; --space-8: 2rem; --space-10: 2.5rem; --space-12: 3rem; --space-16: 4rem; --space-20: 5rem; /* ── Radios ── */ --radius-sm: 4px; --radius-md: 8px; --radius-lg: 16px; --radius-full: 9999px; /* ── Sombras ── */ --shadow-sm: 0 1px 3px hsl(220 15% 12% / 0.08), 0 1px 2px hsl(220 15% 12% / 0.04); --shadow-md: 0 4px 12px hsl(220 15% 12% / 0.10), 0 2px 4px hsl(220 15% 12% / 0.06); --shadow-lg: 0 8px 24px hsl(220 15% 12% / 0.12), 0 4px 8px hsl(220 15% 12% / 0.06); /* ── Transiciones ── */ --transition-fast: 150ms ease; --transition-base: 250ms ease; --transition-slow: 400ms ease; /* ── Anchos de contenedor ── */ --container-sm: 640px; --container-md: 768px; --container-lg: 1024px; --container-xl: 1280px; }
